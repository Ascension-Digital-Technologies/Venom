cmake_minimum_required(VERSION 3.20)
project(venom_v4m5_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
set(V4M5_COMMON frontend.cpp ../v4m3/vm.cpp ../../src/package/hash.cpp)
add_executable(venom_v4m5_js cli.cpp ${V4M5_COMMON})
add_executable(venom_v4m5_tests tests.cpp ${V4M5_COMMON})
foreach(target venom_v4m5_js venom_v4m5_tests)
  target_include_directories(${target} PRIVATE ../../src)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME venom_v4m5_structured_source COMMAND venom_v4m5_tests)
