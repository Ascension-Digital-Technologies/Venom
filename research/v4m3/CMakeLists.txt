cmake_minimum_required(VERSION 3.20)
project(venom_v4m3_research LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
add_executable(venom_v4m3_tests tests.cpp vm.cpp ../../src/package/hash.cpp)
target_include_directories(venom_v4m3_tests PRIVATE ../../src)
if(MSVC)
  target_compile_options(venom_v4m3_tests PRIVATE /W4 /WX)
else()
  target_compile_options(venom_v4m3_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME venom_v4m3_differential COMMAND venom_v4m3_tests)
